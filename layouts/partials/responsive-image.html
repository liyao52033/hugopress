{{/* 响应式图片生成 Partial 用法：{{ partial "responsive-image.html" (dict "src" "images/photo.jpg" "breakpoints" (dict "480" "480x" "800" "800x" "1200" "1200x" "1920" "1920x") "quality" 70 "format" "webp") }} 参数说明： - src: 图片路径（相对于 assets 目录） - breakpoints: 断点配置，key 为宽度描述，value 为 Hugo 尺寸格式 - quality: 图片质量（默认 70） - format: 输出格式（默认 webp） 返回值：包含以下字段的字典 - src: 默认图片 URL（最大尺寸或原图） - srcset: srcset 属性值 - isLocal: 是否为本地资源（可处理） - width: 原图宽度 - height: 原图高度 */}} {{ $src := .src }} {{ $breakpoints := .breakpoints | default (dict "480" "480x" "800" "800x" "1200" "1200x" "1920" "1920x") }} {{ $quality := .quality | default 70 }} {{ $format := .format | default "webp" }} {{ $result := dict "src" $src "srcset" "" "isLocal" false "width" 0 "height" 0 }} {{ if not (hasPrefix $src "http") }} {{ $resource := resources.Get $src }} {{ if $resource }} {{ $result = dict "src" "" "srcset" "" "isLocal" true "width" $resource.Width "height" $resource.Height }} {{ $srcsetParts := slice }} {{ $maxSizeResource := "" }} {{ $maxWidth := 0 }} {{ range $width, $size := $breakpoints }} {{/* 只处理小于等于原图宽度的尺寸 */}} {{ $targetWidth := int $width }} {{ if le $targetWidth $resource.Width }} {{ $resized := $resource.Resize (printf "%s %s q%d" $size $format $quality) }} {{ if $resized }} {{ $srcsetParts = $srcsetParts | append (printf "%s %sw" $resized.RelPermalink $width) }} {{/* 选择最大的有效尺寸作为默认 src */}} {{ if gt $targetWidth $maxWidth }} {{ $maxWidth = $targetWidth }} {{ $maxSizeResource = $resized }} {{ end }} {{ end }} {{ end }} {{ end }} {{/* 使用最大尺寸作为默认 src */}} {{ if $maxSizeResource }} {{ $result = dict "src" $maxSizeResource.RelPermalink "srcset" (delimit $srcsetParts ", ") "isLocal" true "width" $resource.Width "height" $resource.Height }} {{ else }} {{/* 如果没有成功生成任何尺寸，使用原图优化版本 */}} {{ $optimized := $resource.Filter (images.Process (printf "format %s q%d" $format $quality)) }} {{ $result = dict "src" $optimized.RelPermalink "srcset" "" "isLocal" true "width" $resource.Width "height" $resource.Height }} {{ end }} {{ else }} {{/* static 目录的文件 */}} {{ $result = dict "src" ($src | relURL) "srcset" "" "isLocal" false "width" 0 "height" 0 }} {{ end }} {{ else }} {{/* 远程 URL */}} {{ $result = dict "src" $src "srcset" "" "isLocal" false "width" 0 "height" 0 }} {{ end }} {{ return $result }}
